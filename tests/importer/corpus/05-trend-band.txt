//@version=5
indicator("Trend band", overlay = true)

factor = input.float(3.0, "Factor", step = 0.1)
atrPeriod = input.int(10, "ATR period")

[line, direction] = ta.supertrend(factor, atrPeriod)

bodyMid = plot((open + close) / 2, "Body", display = display.none)
upLine = plot(direction < 0 ? line : na, "Up trend", color = color.green, style = plot.style_linebr)
downLine = plot(direction < 0 ? na : line, "Down trend", color = color.red, style = plot.style_linebr)

fill(bodyMid, upLine, color = color.new(color.green, 90))
fill(bodyMid, downLine, color = color.new(color.red, 90))

flipped = direction != direction[1]
plotshape(flipped and direction < 0, title = "Long", style = shape.labelup, location = location.belowbar, color = color.green, text = "Up", textcolor = color.white, size = size.small)
plotshape(flipped and direction > 0, title = "Short", style = shape.labeldown, location = location.abovebar, color = color.red, text = "Down", textcolor = color.white, size = size.small)
