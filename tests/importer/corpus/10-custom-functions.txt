//@version=6
indicator("Custom functions")

mode = input.string("Weighted", "Average", options = ["Simple", "Weighted", "Hull"])
length = input.int(21, "Length")
step = input.int(3, "Spacing")

// A moving average chosen by name.
f_average(src, len) =>
    float result = na
    switch mode
        "Simple" => result := ta.sma(src, len)
        "Weighted" => result := ta.wma(src, len)
        => result := ta.hma(src, len)
    result

f_range(len) => ta.highest(high, len) - ta.lowest(low, len)

f_ladder(base, gap) =>
    total = 0.0
    for i = 1 to 3
        total := total + base + i * gap
    total / 3

average = f_average(close, length)
spread = f_range(length)

// A label for the regime, written by a switch with no subject.
regime = 0
switch
    close > average + spread / 2 => regime := 2
    close > average => regime := 1
    => regime := -1

count = 0
i = 0
while i < 5 and close[i] > average
    count := count + 1
    i := i + 1

if regime > 0
    level = average + spread
    count := count + (close > level ? 1 : 0)

plot(average, "Average", color = color.blue)
plot(f_ladder(average, step), "Ladder", color = color.gray)
plot(regime, "Regime", color = color.orange, style = plot.style_stepline)
plot(count, "Run", color = color.fuchsia, style = plot.style_histogram)
