//@version=5
strategy("Channel breakout", overlay = true, initial_capital = 200000, default_qty_type = strategy.fixed, default_qty_value = 5, commission_type = strategy.commission.cash_per_order, commission_value = 20, slippage = 2)

entryLength = input.int(20, "Entry channel")
exitLength = input.int(10, "Exit channel")

upperEntry = ta.highest(high, entryLength)[1]
lowerEntry = ta.lowest(low, entryLength)[1]
upperExit = ta.highest(high, exitLength)[1]
lowerExit = ta.lowest(low, exitLength)[1]

if close > upperEntry
    strategy.entry("Long", strategy.long)
if close < lowerEntry
    strategy.entry("Short", strategy.short)

strategy.exit("Long exit", "Long", stop = lowerExit)
strategy.exit("Short exit", "Short", stop = upperExit)

plot(upperEntry, "Upper entry", color = color.green)
plot(lowerEntry, "Lower entry", color = color.red)
plot(strategy.position_size > 0 ? lowerExit : na, "Long stop", color = color.orange, style = plot.style_linebr)
plot(strategy.position_size < 0 ? upperExit : na, "Short stop", color = color.orange, style = plot.style_linebr)
